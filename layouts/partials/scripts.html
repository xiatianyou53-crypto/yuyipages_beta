(function () { // 移动端菜单开关 var toggle = document.getElementById('navToggle'); var nav = document.getElementById('mainNav'); if (toggle && nav) { toggle.addEventListener('click', function () { var open = nav.classList.toggle('is-open'); toggle.setAttribute('aria-expanded', open ? 'true' : 'false'); }); } // 首页轮播 var slider = document.getElementById('heroSlider'); if (slider) { var slides = slider.querySelectorAll('.slide'); var dots = slider.querySelectorAll('.dot'); var idx = 0; function go(n) { idx = (n + slides.length) % slides.length; slides.forEach(function (s, i) { s.classList.toggle('is-active', i === idx); }); dots.forEach(function (d, i) { d.classList.toggle('is-active', i === idx); }); } slider.querySelectorAll('.slide-btn').forEach(function (b) { b.addEventListener('click', function () { go(idx + parseInt(b.dataset.step, 10)); }); }); dots.forEach(function (d) { d.addEventListener('click', function () { go(parseInt(d.dataset.index, 10)); }); }); setInterval(function () { go(idx + 1); }, 5000); } })();
